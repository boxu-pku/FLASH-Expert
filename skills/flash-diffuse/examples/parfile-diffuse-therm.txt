# === Diffuse 主控 ===
useDiffuse       = .true.
useDiffuseTherm  = .true.
diff_useEleCond  = .true.         # 启用电子热传导（3T）
# diff_useIonCond = .false.       # 离子热传导（HEDP 通常关，离子温度变化慢）
diffusion_cutoff_density = 1.e-5  # HEDP 推荐（默认 1e-30 太松，真空区会奇异）

# === θ + flux limiter（LaserSlab 经典配置）===
diff_thetaImplct = 1.0            # 后向 Euler，无条件稳定
                                    # ★ 为什么不用 0.5：HEDP 陡梯度 CN 会震荡
diff_eleFlMode   = "fl_larsen"    # Larsen 平滑过渡（HEDP 标准）
                                    # ★ 必须显式设！默认 fl_none 危险（超光速）
diff_eleFlCoef   = 0.06           # 6% 自由流（HEDP 经验值）
                                    # ★ 为什么 0.06：Spitzer 高估低密度区电导率

# === 边界条件（6 面，2D R-Z）===
diff_eleXlBoundaryType = "neumann"   # 左（r=0 轴）零梯度（轴对称）
diff_eleXrBoundaryType = "neumann"   # 右（出靶面）
diff_eleYlBoundaryType = "vacuum"    # ★ 底（激光进入方向，真空让热逃）
diff_eleYrBoundaryType = "neumann"   # 顶（出靶面）
diff_eleZlBoundaryType = "neumann"   # 3D 用（2D 忽略）
diff_eleZrBoundaryType = "neumann"

# === dt 限制 ===
# ★ HEDP 隐式不需要 dt 限制，禁用避免步数过多
dt_diff_factor = 1.0e+100

# === HYPRE ===
gr_hypreSolverType     = "HYPRE_AMG"   # 直接 BoomerAMG（LaserSlab 标准）
                                          # ★ 为什么不用 PCG：超大问题避免双重存储
gr_hyprePCType         = "HYPRE_AMG"   # （AMG 独立时 PCType 忽略）
gr_hypreMaxIter        = 500            # 默认（HEDP 通常够）
gr_hypreRelTol         = 1.0e-8         # 默认（HEDP 物理精度够）
gr_hyprePrintSolveInfo = .true.         # ★ 调试用：看 num_iterations / final_res_norm
gr_hypreFloor          = 1.0e-12        # 防 HYPRE 输出小负值
gr_hypreUseFloor       = .true.
