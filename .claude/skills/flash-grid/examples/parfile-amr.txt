# ===== AMR 层级 =====

nblockz = 1                        # KAXIS（2D 算例设 1）

# ===== 细化变量 1：密度（最常用）=====
refine_var_1 = "dens"              # ★ 变量名（字符串，大小写敏感）
refine_cutoff_1 = 0.2              # 误差 > 0.2 → 细化
derefine_cutoff_1 = 0.05           # 误差 < 0.05 → 粗化
