# ============================================================
# 配置 1：6 群（LaserSlab 标准，5-decade 对数）

# 群范围：0.1 eV - 100 keV

rt_mgdNumGroups = 6

rt_mgdBounds_2 = 1.e0
rt_mgdBounds_3 = 1.e1
rt_mgdBounds_4 = 1.e2
rt_mgdBounds_5 = 1.e3
rt_mgdBounds_6 = 1.e4
rt_mgdBounds_7 = 1.e5


# ============================================================
# 配置 2：1 群（gray，测试/教学用）
# ============================================================
# setup 命令：mgd_meshgroups=1
# 适合：单元测试、教学、验证 MGD 基本工作

rt_mgdNumGroups = 1
rt_mgdBounds_1 = 1.e-3
rt_mgdBounds_2 = 1.e5


# ============================================================
# 配置 3：10 群（X 射线带加密）
# ============================================================
# setup 命令：mgd_meshgroups=10
# 适合：需要精细 X 射线能谱演化的算例

rt_mgdNumGroups = 10
# 低能区粗，高能区（X 射线）细
rt_mgdBounds_1 = 1.e0
rt_mgdBounds_2 = 1.e1
rt_mgdBounds_3 = 1.e2
rt_mgdBounds_4 = 5.e2
rt_mgdBounds_5 = 1.e3
rt_mgdBounds_6 = 2.e3
rt_mgdBounds_7 = 3.e3
rt_mgdBounds_8 = 5.e3
rt_mgdBounds_9 = 7.e3
rt_mgdBounds_10 = 1.e4
rt_mgdBounds_11 = 1.e5


# ============================================================
# 配置 4：70 群（精细谱，X 射线诊断）
# ============================================================
# setup 命令：mgd_meshgroups=70
# 适合：需要详细 X 射线谱诊断的算例
# 注意：必须用匹配的 IONMIX 表（100 群表取子集）

rt_mgdNumGroups = 70
# 70 个 rt_mgdBounds_* 要手动写
# 建议从 IONMIX 表（如 polystyrene-imx-008.cn4）的群边界取子集
# rt_mgdBounds_1 = ...
# rt_mgdBounds_2 = ...
# ...一直到 rt_mgdBounds_71


# ============================================================
# 选择群数的决策（按 L-013）
#
# 群数选择：
# - 1 群（gray）：教学/快速测试，物理粗糙
# - 6 群（LaserSlab 标准）：HEDP 推荐，覆盖 5-decade
# - 10-30 群：需要 X 射线诊断精度
# - 70+ 群：研究级辐射谱演化，需要匹配的 IONMIX 表
#
# 注意：群数越多 →
# - 内存：每个 MGDR_VAR 都占 NUNK_VARS 一个槽
# - 时间：每群一次 HYPRE 求解（线性增长）
# - IONMIX 表必须匹配（手动对齐）
# ============================================================
