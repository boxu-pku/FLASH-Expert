### Setup 命令必须和原 run 完全一致！###
#
# 必须相同的选项：
#   -2d / -3d            # 维度
#   -nxb=N -nyb=N -nzb=N # 块大小（NXB/NYB/NZB 是编译时常量）
#   +cylindrical / ...   # 几何
#   +laser +mgd +uhd3t   # 物理 setup 变量
#   species=...          # 物质
#
# 否则 io_checkBlockShape.F90:L39-L41 会 abort：
#   "NXB in checkpoint does not match this executable!"

### Restart 续跑步骤 ###

# 1. 用和原 run 完全一致的 setup 命令（仅 -objdir 和 -parfile 不同）：

#    make -j4

# 3. 拷贝原 run 的 chk 文件过来：
#    cp ../LaserSlab/lasslab_hdf5_chk_0015 .
#    # 注意：outputSplitNum > 1 时需拷所有 split 文件
#
# 4. 跑：
#    ./flash4 -par_file=flash.par
#
# 5. 验证：日志开头应输出 "Read checkpoint file lasslab_hdf5_chk_0015"

### 自动 restart（替代方案）###

# 如果嫌手动改 parfile 麻烦，可用 appendParRestart = .true.：
# 每次写 chk 后会自动追加到 parfile：
#   ### BEGIN written by [IO_writeCheckpoint] ###
#   restart = .true.
#   checkpointFileNumber = 0015
#   plotFileNumber = 0007
#   ### END written by [IO_writeCheckpoint] ###
# 下次跑同一个 parfile 就会自动 restart

appendParRestart = .false.              # 默认 false（用户希望 parfile 不被修改）
